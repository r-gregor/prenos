" --- CHANGE 'echo ' into 'printf ' ---
" pp: echo "..."      --> printf "[i] ... \n"
" pf: echo "..."      --> printf " ... \n"
" pe: echo -e "...\n" --> printf " ... \n"
nnoremap <LEADER>pp ^ciwprintf<ESC>f"a[i] <ESC>f"i\n<ESC>j
nnoremap <LEADER>pf ^ciwprintf<ESC>$i\n<ESC>j
nnoremap <LEADER>pe ^vt"cprintf <ESC>j
